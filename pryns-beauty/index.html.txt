        // Lógica del Carrito de Compras
        function addToCart(productId) {
            const product = products.find(p => p.id === productId);
            if (!product) return;

            const existing = cart.find(item => item.id === productId);
            if (existing) {
                existing.quantity += 1;
            } else {
                cart.push({ ...product, quantity: 1 });
            }

            updateCartUI();
            showToast(`" ${product.name} " agregado al pedido`);
        }

        function removeFromCart(productId) {
            cart = cart.filter(item => item.id !== productId);
            updateCartUI();
        }

        function updateQuantity(productId, delta) {
            const item = cart.find(i => i.id === productId);
            if (!item) return;

            item.quantity += delta;
            if (item.quantity <= 0) {
                removeFromCart(productId);
            } else {
                updateCartUI();
            }
        }

        function updateCartUI() {
            const totalCount = cart.reduce((sum, item) => sum + item.quantity, 1 ? 0 : 0);
            const totalItems = cart.reduce((sum, item) => sum + item.quantity, 0);
            document.getElementById('cartBadge').innerText = totalItems;

            const isWholesale = totalItems >= 3;

            let totalDetal = 0;
            let totalMayor = 0;

            const list = document.getElementById('cartItemsList');
            if (cart.length === 0) {
                list.innerHTML = `
                    <div class="text-center py-12 text-gray-500">
                        <i class="fa-solid fa-bag-shopping text-4xl mb-3 text-obsidian-700"></i>
                        <p class="text-xs">Tu carrito está vacío en este momento.</p>
                    </div>
                `;
            } else {
                list.innerHTML = cart.map(item => {
                    const priceToUse = isWholesale ? item.baseMayor : item.baseDetal;
                    totalDetal += item.baseDetal * item.quantity;
                    totalMayor += item.baseMayor * item.quantity;

                    return `
                        <div class="flex items-center justify-between bg-obsidian-950 p-3.5 rounded-2xl border border-obsidian-800">
                            <img src="${item.image}" alt="${item.name}" class="w-14 h-14 object-cover rounded-xl border border-obsidian-700">
                            <div class="flex-1 mx-3">
                                <h4 class="text-xs font-bold text-white line-clamp-1">${item.name}</h4>
                                <span class="text-[10px] text-rosegold-400 font-semibold">$${priceToUse}.00 c/u</span>
                            </div>
                            <div class="flex items-center space-x-2 bg-obsidian-900 border border-obsidian-800 rounded-lg p-1">
                                <button onclick="updateQuantity(${item.id}, -1)" class="w-5 h-5 flex items-center justify-center text-gray-400 hover:text-white text-xs">-</button>
                                <span class="text-xs font-bold text-white px-1">${item.quantity}</span>
                                <button onclick="updateQuantity(${item.id}, 1)" class="w-5 h-5 flex items-center justify-center text-gray-400 hover:text-white text-xs">+</button>
                            </div>
                        </div>
                    `;
                }).join('');
            }

            document.getElementById('cartTotalDetal').innerText = `$${totalDetal}.00`;
            document.getElementById('cartTotalMayor').innerText = `$${totalMayor}.00`;
        }

        function toggleCartModal() {
            const modal = document.getElementById('cartModal');
            const container = document.getElementById('cartContainer');
            const isOpen = !modal.classList.contains('opacity-0');

            if (isOpen) {
                modal.classList.add('opacity-0', 'pointer-events-none');
                container.classList.add('translate-x-full');
            } else {
                modal.classList.remove('opacity-0', 'pointer-events-none');
                container.classList.remove('translate-x-full');
            }
        }

        function sendWhatsAppOrder() {
            if (cart.length === 0) {
                showToast('Agrega fragancias al carrito antes de hacer el pedido');
                return;
            }

            const totalItems = cart.reduce((sum, item) => sum + item.quantity, 0);
            const isWholesale = totalItems >= 3;
            
            let message = `✨ *NUEVA ORDEN DE COMPRA - PRYNS BEAUTY* ✨\n\n`;
            message += `Hola, me gustaría solicitar los siguientes productos de su catálogo:\n\n`;

            let grandTotal = 0;
            cart.forEach((item, index) => {
                const unitPrice = isWholesale ? item.baseMayor : item.baseDetal;
                const subtotal = unitPrice * item.quantity;
                grandTotal += subtotal;
                message += `${index + 1}. *${item.name}* (${item.brand})\n`;
                message += `   • Cantidad: ${item.quantity}\n`;
                message += `   • Precio Unitario: $${unitPrice}.00\n`;
                message += `   • Subtotal: $${subtotal}.00\n\n`;
            });

            message += `----------------------------------------\n`;
            message += `📦 *Total de Piezas:* ${totalItems}\n`;
            message += `🏷️ *Modalidad:* ${isWholesale ? 'Mayorista (Aplica Descuento)' : 'Detal'}\n`;
            message += `💰 *TOTAL A PAGAR:* $${grandTotal}.00 USD\n\n`;
            message += `Por favor confirmen disponibilidad y métodos de pago. ¡Muchas gracias!`;

            const encodedMessage = encodeURIComponent(message);
            const whatsappUrl = `https://wa.me/584122588332?text=${encodedMessage}`;
            window.open(whatsappUrl, '_blank');
        }
